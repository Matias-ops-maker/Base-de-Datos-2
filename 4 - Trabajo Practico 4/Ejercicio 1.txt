  { _id: 1, nombre: "Proyecto A", participantes: [1, 2] },
  { _id: 2, nombre: "Proyecto B", participantes: [3] }
]);

db.tareas.insertMany([
  { _id: 1, nombre: "Diseño", estado: "Pendiente", proyecto_id: 1, asignada_a: 1 },
  { _id: 2, nombre: "Implementación", estado: "En Progreso", proyecto_id: 1, asignada_a: 2 },
  { _id: 3, nombre: "Testing", estado: "Completada", proyecto_id: 2, asignada_a: 3 }
]);

// CONSULTAS
db.proyectos.aggregate([
  {
    $lookup: {
      from: "personas",
      localField: "participantes",
      foreignField: "_id",
      as: "participantes_info"
    }
  },
  {
    $project: {
      nombre: 1,
      participantes: "$participantes_info.nombre"
    }
  }
]);

db.tareas.find({ asignada_a: 1 });
db.proyectos.find({ participantes: 2 });
