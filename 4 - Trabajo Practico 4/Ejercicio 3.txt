db.contactos.insertMany([
  { persona_id: 1, contacto_id: 2 },
  { persona_id: 2, contacto_id: 3 }
]);

// CONSULTAS
db.trabajos.aggregate([
  { $match: { empresa_id: 1 } },
  { $lookup: {
      from: "personas",
      localField: "persona_id",
      foreignField: "_id",
      as: "persona"
  }},
  { $unwind: "$persona" },
  { $project: { nombre: "$persona.nombre" } }
]);

db.persona_habilidad.aggregate([
  { $match: { habilidad_id: 1 } },
  { $lookup: {
      from: "personas",
      localField: "persona_id",
      foreignField: "_id",
      as: "persona"
  }},
  { $unwind: "$persona" },
  { $project: { nombre: "$persona.nombre" } }
]);

var primerGrado = db.contactos.find({ persona_id: 1 }).map(x => x.contacto_id);
db.contactos.find({ persona_id: { $in: primerGrado }, contacto_id: { $ne: 1 } });
